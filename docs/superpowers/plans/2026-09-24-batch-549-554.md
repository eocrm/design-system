// Initial focus belongs to the OPEN transition only. isTop also flips
// false → true when a nested overlay (e.g. a Modal opened from this drawer)
// closes; re-running here would queue a focus that lands AFTER the nested
// overlay restored focus to its opener / returnFocusRef (#551).
const initialFocusPendingRef = useRef(true);
useLayoutEffect(() => {
  if (!ctx.open) {
    initialFocusPendingRef.current = true;
    return;
  }
  if (!ctx.isTop || !initialFocusPendingRef.current) return;
  initialFocusPendingRef.current = false;
  queueMicrotask(() => {
    const target = ctx.initialFocusRef?.current ?? ctx.contentRef.current;
    target?.focus({ preventScroll: true });
  });
}, [ctx.open, ctx.isTop, ctx.initialFocusRef, ctx.contentRef]);

```tsx
/**
 * Returns focus when the popover closes (#552). Popover itself only refocuses
 * the trigger on Escape / Popover.Close; a Confirm or Cancel click unmounts
 * the focused button and would drop focus to <body>. Runs in a layout effect
 * on the open → closed commit, after React has removed the panel.
 */
function FocusReturn({ returnFocusRef }: { returnFocusRef?: RefObject<HTMLElement | null> }) {
  const ctx = usePopoverContext(/* check the hook's signature */);
  const prevOpenRef = useRef(ctx.open);
  useLayoutEffect(() => {
    const wasOpen = prevOpenRef.current;
    prevOpenRef.current = ctx.open;
    if (ctx.open || !wasOpen) return;
    const trigger = ctx.triggerRef.current;
    const active = document.activeElement;
    // An outside click onto a real control already moved focus — respect it.
    // Focus "was lost" when it sits on <body>, on a detached node, or on the
    // trigger (Popover's Escape path refocuses the trigger before we run).
    const lost = !active || active === document.body || !active.isConnected || active === trigger;
    if (!lost) return;
    const requested = returnFocusRef?.current ?? null;
    if (requested && requested.isConnected) {
      requested.focus({ preventScroll: true });
      // #553: the named target is often far from where the user was.
      requested.scrollIntoView?.({ block: 'nearest' });
      return;
    }
    if (trigger?.isConnected) trigger.focus({ preventScroll: true });
  }, [ctx.open, ctx.triggerRef, returnFocusRef]);
  return null;
}
```

```tsx
const ANNOUNCE_DELAY_MS = 50;

export const LiveRegion = forwardRef<HTMLSpanElement, LiveRegionProps>(function LiveRegion(
  { children, politeness = 'polite', announceKey, ...props },
  ref,
) {
  const [shown, setShown] = useState<ReactNode>(null);
  // Strings / numbers compare by value so a parent re-render with the same
  // text does not re-announce; anything else compares by identity (JSDoc
  // steers consumers to strings).
  const messageKey =
    typeof children === 'string' || typeof children === 'number' ? String(children) : children;
  const latest = useRef(children);
  latest.current = children;

  useEffect(() => {
    // Clear, then write in a later task: the region goes empty → text, which
    // screen readers treat as a fresh change — that is what makes a mount-
    // with-message and an identical re-announce (announceKey) both fire.
    setShown(null);
    const id = setTimeout(() => setShown(latest.current), ANNOUNCE_DELAY_MS);
    return () => clearTimeout(id);
  }, [messageKey, announceKey]);

  // {...props} first so role / aria-live / aria-atomic always win (Pattern B).
  return (
    <VisuallyHidden
      {...props}
      ref={ref}
      role={politeness === 'assertive' ? 'alert' : 'status'}
      aria-live={politeness}
      aria-atomic="true"
    >
      {shown}
    </VisuallyHidden>
  );
});
```

(`latest.current = children` during render is the established ref-latest pattern; if the repo's eslint `react-hooks` rules flag it, move the assignment into a `useLayoutEffect` without deps.)

Full JSDoc: description; props docs (children: "The message. Pass a string — …"; politeness options with guidance; announceKey: "Change it (e.g. a counter) to announce the same message again."); `@example`s: two-factor success status, assertive form-level error, `announceKey` repeat save; `@remarks` anti-patterns from the spec:
- ❌ Announcing text that is already visible and focused (reads twice).
- ❌ Mounting it conditionally (`{msg && <LiveRegion>…}`) — keep it mounted; pass an empty message instead. (It still announces on mount, but a region that mounts and unmounts per message is churn.)
- ❌ For a library component's own transient state — components own their regions (Hard rule 10); this is for consumer-level outcomes.
- ❌ `assertive` for routine success — reserve it for errors needing immediate attention.
- ❌ JSX children when a string would do — element children re-announce on every parent render.
